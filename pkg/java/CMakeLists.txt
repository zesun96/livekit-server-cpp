cmake_minimum_required(VERSION 3.24)
project(livekit_server_java VERSION 0.1.0 LANGUAGES CXX)
enable_testing()
find_package(Protobuf CONFIG REQUIRED)

set(JAVA_JNI_INCLUDE_DIR "" CACHE PATH "Directory containing jni.h and platform jni_md.h")
if(JAVA_JNI_INCLUDE_DIR)
  set(_jni_includes "${JAVA_JNI_INCLUDE_DIR}")
else()
  find_package(JNI REQUIRED COMPONENTS JVM)
  set(_jni_includes ${JNI_INCLUDE_DIRS})
endif()
# Build the SDK in the same tree to keep architecture, runtime and protobuf ABI consistent.
if(NOT TARGET LiveKitServer::livekitserver)
  set(LIVEKIT_SERVER_BUILD_TESTS OFF CACHE BOOL "Build deterministic unit tests")
  set(LIVEKIT_SERVER_BUILD_EXAMPLES OFF CACHE BOOL "Build examples")
  add_subdirectory(../.. sdk)
endif()
set_property(TARGET livekitserver PROPERTY POSITION_INDEPENDENT_CODE ON)
add_library(livekitserver_java SHARED src/main/cpp/livekit_server_jni.cpp)
target_compile_features(livekitserver_java PRIVATE cxx_std_20)
target_include_directories(livekitserver_java PRIVATE ${_jni_includes} ../../src)
target_link_libraries(livekitserver_java PRIVATE LiveKitServer::livekitserver protobuf::libprotobuf)
if(MSVC)
  target_compile_options(livekitserver_java PRIVATE /W4 /permissive- /utf-8)
endif()
install(TARGETS livekitserver_java RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
