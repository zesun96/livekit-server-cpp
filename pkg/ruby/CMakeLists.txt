cmake_minimum_required(VERSION 3.24)
project(livekit_server_ruby VERSION 0.1.0 LANGUAGES CXX)

find_package(Protobuf CONFIG REQUIRED)
if(BUILD_SHARED_LIBS)
  message(FATAL_ERROR "The Ruby bridge embeds a static SDK; configure with BUILD_SHARED_LIBS=OFF")
endif()
if(NOT TARGET LiveKitServer::livekitserver)
  set(LIVEKIT_SERVER_BUILD_TESTS OFF CACHE BOOL "Build deterministic unit tests")
  set(LIVEKIT_SERVER_BUILD_EXAMPLES OFF CACHE BOOL "Build examples")
  add_subdirectory(../.. sdk)
endif()
set_property(TARGET livekitserver PROPERTY POSITION_INDEPENDENT_CODE ON)
add_library(livekitserver_ruby SHARED src/livekit_server_ruby.cpp)
target_compile_features(livekitserver_ruby PRIVATE cxx_std_20)
target_include_directories(livekitserver_ruby PRIVATE ../../src)
target_link_libraries(livekitserver_ruby PRIVATE LiveKitServer::livekitserver protobuf::libprotobuf)
if(MSVC)
  target_compile_options(livekitserver_ruby PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(livekitserver_ruby PRIVATE -Wall -Wextra -Wpedantic)
endif()
